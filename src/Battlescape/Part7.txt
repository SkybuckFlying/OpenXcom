destructor TBattlescapeState.Destroy;
begin
  if FSave.GetAmbientSound <> -1 then
    FMod.GetSoundByDepth(FSave.GetDepth, FSave.GetAmbientSound).StopLoop;
  FAnimTimer.Free;
  FGameTimer.Free;
  FBattleGame.Free;
  FPopups.Free;
  inherited;
end;

procedure TBattlescapeState.Init;
begin
  if FSave.GetAmbientSound <> -1 then
    FMod.GetSoundByDepth(FSave.GetDepth, FSave.GetAmbientSound).Loop;
  FGame.SetVolume(Options.SoundVolume, Options.MusicVolume, Options.UIVolume);
  inherited Init;
  FAnimTimer.Start;
  FGameTimer.Start;
  FMap.SetFocus(True);
  FMap.CacheUnits;
  FMap.Draw;
  FBattleGame.Init;
  UpdateSoldierInfo;
  if FFirstInit then
  begin
    if not PlayableUnitSelected then
      SelectNextPlayerUnit(False, False, False, True);
    if PlayableUnitSelected then
    begin
      FBattleGame.SetupCursor;
      FMap.GetCamera.CenterOnPosition(FSave.GetSelectedUnit.GetPosition);
    end;
    FFirstInit := False;
  end;
  FBtnReserveKneel.Toggle(FSave.GetKneelReserved);
  FBattleGame.SetKneelReserved(FSave.GetKneelReserved);
  if FAutosave then
  begin
    FAutosave := False;
    if FGame.GetSavedGame.IsIronman then
      FGame.PushState(TSaveGameState.Create(OPT_BATTLESCAPE, SAVE_IRONMAN, FPalette))
    else if Options.Autosave then
      FGame.PushState(TSaveGameState.Create(OPT_BATTLESCAPE, SAVE_AUTO_BATTLESCAPE, FPalette));
  end;
end;

procedure TBattlescapeState.Think;
begin
  if FGameTimer.IsRunning then
  begin
    if FPopups.Count > 0 then
    begin
      FGame.PushState(FPopups[0]);
      FPopups.Delete(0);
      Exit;
    end;
    inherited Think;
    FBattleGame.Think;
    FAnimTimer.Think(Self, 0);
    FGameTimer.Think(Self, 0);
  end;
end;

procedure TBattlescapeState.Handle(AAction: TAction);
begin
  if FFirstInit then Exit;
  if FGame.GetCursor.Visible or
     ((AAction.GetDetails.typ = SDL_MOUSEBUTTONDOWN) or
      (AAction.GetDetails.typ = SDL_MOUSEBUTTONUP)) and
     (AAction.GetDetails.button.button = SDL_BUTTON_RIGHT) then
  begin
    inherited Handle(AAction);

    if (not Options.TouchEnabled) and FIsMouseScrolling and (not Options.BattleDragScrollInvert) then
      FMap.SetSelectorPosition((FCursorPosition.X - FGame.GetScreen.CursorLeftBlackBand) / AAction.GetXScale,
                               (FCursorPosition.Y - FGame.GetScreen.CursorTopBlackBand) / AAction.GetYScale);

    if AAction.GetDetails.typ = SDL_MOUSEBUTTONDOWN then
    begin
      if AAction.GetDetails.button.button = SDL_BUTTON_X1 then BtnNextSoldierClick(AAction);
      if AAction.GetDetails.button.button = SDL_BUTTON_X2 then BtnPrevSoldierClick(AAction);
    end;

    if (AAction.GetDetails.typ = SDL_KEYDOWN) and Options.Debug then
    begin
      if (AAction.GetDetails.key.keysym.sym = SDLK_d) and ((SDL_GetModState and KMOD_CTRL) <> 0) then
      begin
        FSave.SetDebugMode;
        Debug('Debug Mode');
      end
      else if FSave.GetDebugMode and (AAction.GetDetails.key.keysym.sym = SDLK_v) and ((SDL_GetModState and KMOD_CTRL) <> 0) then
      begin
        Debug('Resetting tile visibility');
        FSave.ResetTiles;
      end
      else if FSave.GetDebugMode and (AAction.GetDetails.key.keysym.sym = SDLK_k) and ((SDL_GetModState and KMOD_CTRL) <> 0) then
      begin
        Debug('Influenza bacterium dispersed');
        for var Unit in FSave.GetUnits do
          if (Unit.GetOriginalFaction = FACTION_HOSTILE) and (not Unit.IsOut) then
            Unit.Damage(Position(0,0,0), 1000, DT_AP, True);
        FBattleGame.CheckForCasualties(nil, nil, True, False);
        FBattleGame.HandleState;
      end
      else if FSave.GetDebugMode and (AAction.GetDetails.key.keysym.sym = SDLK_j) and ((SDL_GetModState and KMOD_CTRL) <> 0) then
      begin
        Debug('Deploying Celine Dion album');
        for var Unit in FSave.GetUnits do
          if (Unit.GetOriginalFaction = FACTION_HOSTILE) and (not Unit.IsOut) then
            Unit.Damage(Position(0,0,0), 1000, DT_STUN, True);
        FBattleGame.CheckForCasualties(nil, nil, True, False);
        FBattleGame.HandleState;
      end
      else if FSave.GetDebugMode and (AAction.GetDetails.key.keysym.sym = SDLK_w) and ((SDL_GetModState and KMOD_CTRL) <> 0) then
      begin
        Debug('Beam me up Scotty');
        var Unit := FSave.GetSelectedUnit;
        var NewPos: TPosition;
        FMap.GetSelectorPosition(NewPos);
        if (Unit <> nil) and (NewPos.X >= 0) then
        begin
          Unit.GetTile.SetUnit(nil);
          Unit.SetPosition(NewPos);
          FSave.GetTile(NewPos).SetUnit(Unit);
          FSave.GetTileEngine.CalculateUnitLighting;
          FBattleGame.HandleState;
        end;
      end
      else if AAction.GetDetails.key.keysym.sym = SDLK_F11 then SaveVoxelMap
      else if (AAction.GetDetails.key.keysym.sym = SDLK_F9) and Options.TraceAI then SaveAIMap;
    end;

    // Quick save/load
    if not FGame.GetSavedGame.IsIronman then
    begin
      if AAction.GetDetails.key.keysym.sym = Options.KeyQuickSave then
        FGame.PushState(TSaveGameState.Create(OPT_BATTLESCAPE, SAVE_QUICK, FPalette))
      else if AAction.GetDetails.key.keysym.sym = Options.KeyQuickLoad then
        FGame.PushState(TLoadGameState.Create(OPT_BATTLESCAPE, SAVE_QUICK, FPalette));
    end;

    if AAction.GetDetails.key.keysym.sym = Options.KeyBattleVoxelView then
      SaveVoxelView;
  end;
end;

procedure TBattlescapeState.Resize(var dX, dY: Integer);
var
  Divisor, I: Integer;
  PixelRatioY: Double;
begin
  dX := Options.BaseXResolution;
  dY := Options.BaseYResolution;
  Divisor := 1;
  PixelRatioY := 1.0;
  if Options.NonSquarePixelRatio then PixelRatioY := 1.2;
  case Options.BattlescapeScale of
    SCALE_SCREEN_DIV_3: Divisor := 3;
    SCALE_SCREEN_DIV_2: Divisor := 2;
    SCALE_SCREEN: ;
    else Exit;
  end;
  Options.BaseXResolution := Max(Screen.ORIGINAL_WIDTH, Options.DisplayWidth div Divisor);
  Options.BaseYResolution := Max(Screen.ORIGINAL_HEIGHT, Round(Options.DisplayHeight / PixelRatioY / Divisor));
  dX := Options.BaseXResolution - dX;
  dY := Options.BaseYResolution - dY;
  FMap.Width := Options.BaseXResolution;
  FMap.Height := Options.BaseYResolution;
  FMap.GetCamera.Resize;
  FMap.GetCamera.JumpXY(dX div 2, dY div 2);
  for I := 0 to FSurfaces.Count - 1 do
  begin
    if (FSurfaces[I] <> FMap) and (FSurfaces[I] <> FBtnPsi) and
       (FSurfaces[I] <> FBtnLaunch) and (FSurfaces[I] <> FTxtDebug) then
    begin
      FSurfaces[I].X := FSurfaces[I].X + dX div 2;
      FSurfaces[I].Y := FSurfaces[I].Y + dY;
    end
    else if (FSurfaces[I] <> FMap) and (FSurfaces[I] <> FTxtDebug) then
      FSurfaces[I].X := FSurfaces[I].X + dX;
  end;
end;

procedure TBattlescapeState.MapOver(AAction: TAction);
begin
  if FIsMouseScrolling and (AAction.GetDetails.typ = SDL_MOUSEMOTION) then
  begin
    if (SDL_GetMouseState(0,0) and SDL_BUTTON(Options.BattleDragScrollButton)) = 0 then
    begin
      if (not FMouseMovedOverThreshold) and
         (SDL_GetTicks - FMouseScrollingStartTime <= Cardinal(Options.DragScrollTimeTolerance)) then
        FMap.GetCamera.MapOffset := FMapOffsetBeforeMouseScrolling;
      FIsMouseScrolled := False;
      FIsMouseScrolling := False;
      StopScrolling(AAction);
      Exit;
    end;
    FIsMouseScrolled := True;
    if not Options.TouchEnabled then
    begin
      SDL_EventState(SDL_MOUSEMOTION, SDL_IGNORE);
      SDL_WarpMouse(FGame.GetScreen.Width div 2, FGame.GetScreen.Height div 2 - FMap.IconHeight div 2);
      SDL_EventState(SDL_MOUSEMOTION, SDL_ENABLE);
    end;
    FTotalMouseMoveX := FTotalMouseMoveX + AAction.GetDetails.motion.xrel;
    FTotalMouseMoveY := FTotalMouseMoveY + AAction.GetDetails.motion.yrel;
    if not FMouseMovedOverThreshold then
      FMouseMovedOverThreshold := (Abs(FTotalMouseMoveX) > Options.DragScrollPixelTolerance) or
                                  (Abs(FTotalMouseMoveY) > Options.DragScrollPixelTolerance);
    if Options.BattleDragScrollInvert then
    begin
      FMap.GetCamera.MapOffset := FMapOffsetBeforeMouseScrolling;
      var ScrollX := -Round(FTotalMouseMoveX / AAction.GetXScale);
      var ScrollY := -Round(FTotalMouseMoveY / AAction.GetYScale);
      var Delta2 := FMap.GetCamera.MapOffset;
      FMap.GetCamera.ScrollXY(ScrollX, ScrollY, True);
      Delta2 := FMap.GetCamera.MapOffset - Delta2;
      if (ScrollX <> Delta2.X) or (ScrollY <> Delta2.Y) then
      begin
        FTotalMouseMoveX := -Round(Delta2.X * AAction.GetXScale);
        FTotalMouseMoveY := -Round(Delta2.Y * AAction.GetYScale);
      end;
      if not Options.TouchEnabled then
      begin
        AAction.GetDetails.motion.x := FXBeforeMouseScrolling;
        AAction.GetDetails.motion.y := FYBeforeMouseScrolling;
      end;
      FMap.CursorType := CT_NONE;
    end
    else
    begin
      var Delta := FMap.GetCamera.MapOffset;
      FMap.GetCamera.MapOffset := FMapOffsetBeforeMouseScrolling;
      var ScrollX := Round(FTotalMouseMoveX / AAction.GetXScale);
      var ScrollY := Round(FTotalMouseMoveY / AAction.GetYScale);
      var Delta2 := FMap.GetCamera.MapOffset;
      FMap.GetCamera.ScrollXY(ScrollX, ScrollY, True);
      Delta2 := FMap.GetCamera.MapOffset - Delta2;
      Delta := FMap.GetCamera.MapOffset - Delta;
      if (ScrollX <> Delta2.X) or (ScrollY <> Delta2.Y) then
      begin
        FTotalMouseMoveX := Round(Delta2.X * AAction.GetXScale);
        FTotalMouseMoveY := Round(Delta2.Y * AAction.GetYScale);
      end;
      var BarWidth := FGame.GetScreen.CursorLeftBlackBand;
      var BarHeight := FGame.GetScreen.CursorTopBlackBand;
      var CursorX := FCursorPosition.X + Round(Delta.X * AAction.GetXScale);
      var CursorY := FCursorPosition.Y + Round(Delta.Y * AAction.GetYScale);
      FCursorPosition.X := Clamp(CursorX, BarWidth, FGame.GetScreen.Width - BarWidth - Round(AAction.GetXScale));
      FCursorPosition.Y := Clamp(CursorY, BarHeight, FGame.GetScreen.Height - BarHeight - Round(AAction.GetYScale));
      if not Options.TouchEnabled then
      begin
        AAction.GetDetails.motion.x := FCursorPosition.X;
        AAction.GetDetails.motion.y := FCursorPosition.Y;
      end;
    end;
    FGame.GetCursor.Handle(AAction);
  end;
end;

procedure TBattlescapeState.MapPress(AAction: TAction);
begin
  if FMouseOverIcons then Exit;
  if AAction.GetDetails.button.button = Options.BattleDragScrollButton then
  begin
    FIsMouseScrolling := True;
    FIsMouseScrolled := False;
    SDL_GetMouseState(FXBeforeMouseScrolling, FYBeforeMouseScrolling);
    FMapOffsetBeforeMouseScrolling := FMap.GetCamera.MapOffset;
    if not Options.BattleDragScrollInvert and (FCursorPosition.Z = 0) then
    begin
      FCursorPosition.X := AAction.GetDetails.motion.x;
      FCursorPosition.Y := AAction.GetDetails.motion.y;
      FCursorPosition.Z := 1;
    end;
    FTotalMouseMoveX := 0;
    FTotalMouseMoveY := 0;
    FMouseMovedOverThreshold := False;
    FMouseScrollingStartTime := SDL_GetTicks;
  end;
end;

procedure TBattlescapeState.MapClick(AAction: TAction);
var
  Pos: TPosition;
begin
  if FIsMouseScrolling then
  begin
    if (AAction.GetDetails.button.button <> Options.BattleDragScrollButton) and
       ((SDL_GetMouseState(0,0) and SDL_BUTTON(Options.BattleDragScrollButton)) = 0) then
    begin
      if (not FMouseMovedOverThreshold) and
         (SDL_GetTicks - FMouseScrollingStartTime <= Cardinal(Options.DragScrollTimeTolerance)) then
        FMap.GetCamera.MapOffset := FMapOffsetBeforeMouseScrolling;
      FIsMouseScrolled := False;
      FIsMouseScrolling := False;
      StopScrolling(AAction);
    end;
    if FIsMouseScrolling then
    begin
      if AAction.GetDetails.button.button = Options.BattleDragScrollButton then
      begin
        FIsMouseScrolling := False;
        StopScrolling(AAction);
      end
      else Exit;
      if (not FMouseMovedOverThreshold) and
         (SDL_GetTicks - FMouseScrollingStartTime <= Cardinal(Options.DragScrollTimeTolerance)) then
      begin
        FIsMouseScrolled := False;
        StopScrolling(AAction);
      end;
      if FIsMouseScrolled then Exit;
    end;
  end;

  if AAction.GetDetails.button.button = SDL_BUTTON_RIGHT then
  begin
    if FBattleGame.CancelCurrentAction then Exit;
  end;

  if FMouseOverIcons then Exit;
  if (FMap.CursorType = CT_NONE) or FBattleGame.IsBusy then Exit;

  FMap.GetSelectorPosition(Pos);
  if FSave.GetDebugMode then
    Debug('Clicked ' + Pos.ToString);

  if FSave.GetTile(Pos) <> nil then
  begin
    if (AAction.GetDetails.button.button = SDL_BUTTON_RIGHT) or
       ((AAction.GetDetails.button.button = SDL_BUTTON_LEFT) and ((SDL_GetModState and KMOD_ALT) <> 0)) then
    begin
      if PlayableUnitSelected then
        FBattleGame.SecondaryAction(Pos);
    end
    else if AAction.GetDetails.button.button = SDL_BUTTON_LEFT then
      FBattleGame.PrimaryAction(Pos);
  end;
end;

procedure TBattlescapeState.UpdateSoldierInfo(CheckFOV: Boolean);
var
  Unit: TBattleUnit;
  I, J: Integer;
  LeftHandItem, RightHandItem: TBattleItem;
begin
  Unit := FSave.GetSelectedUnit;
  for I := 0 to VISIBLE_MAX - 1 do
  begin
    FBtnVisibleUnit[I].Visible := False;
    FNumVisibleUnit[I].Visible := False;
    FVisibleUnit[I] := nil;
  end;
  var Playable := FBattleGame.PlayableUnitSelected;
  FRankSurface.Visible := Playable;
  FNumTimeUnits.Visible := Playable;
  FBarTimeUnits.Visible := Playable;
  FNumEnergy.Visible := Playable;
  FBarEnergy.Visible := Playable;
  FNumHealth.Visible := Playable;
  FBarHealth.Visible := Playable;
  FNumMorale.Visible := Playable;
  FBarMorale.Visible := Playable;
  FBtnLeftHand.Visible := Playable;
  FBtnRightHand.Visible := Playable;
  FNumAmmoLeft.Visible := Playable;
  FNumAmmoRight.Visible := Playable;
  if not Playable then
  begin
    FTxtName.Text := '';
    ShowPsiButton(False);
    ToggleKneelButton(nil);
    Exit;
  end;

  FTxtName.Text := Unit.GetName(FGame.GetLanguage, False);
  var Soldier := Unit.GetGeoscapeSoldier;
  if Soldier <> nil then
    FMod.GetSurfaceSet('SMOKE.PCK').GetFrame(20 + Soldier.Rank).Blit(FRankSurface)
  else
    FRankSurface.Clear;

  FNumTimeUnits.Value := Unit.TimeUnits;
  FBarTimeUnits.Max := Unit.BaseStats.TU;
  FBarTimeUnits.Value := Unit.TimeUnits;
  FNumEnergy.Value := Unit.Energy;
  FBarEnergy.Max := Unit.BaseStats.Stamina;
  FBarEnergy.Value := Unit.Energy;
  FNumHealth.Value := Unit.Health;
  FBarHealth.Max := Unit.BaseStats.Health;
  FBarHealth.Value := Unit.Health;
  FBarHealth.Value2 := Unit.StunLevel;
  FNumMorale.Value := Unit.Morale;
  FBarMorale.Max := 100;
  FBarMorale.Value := Unit.Morale;

  ToggleKneelButton(Unit);

  LeftHandItem := GetLeftHandItem(Unit);
  FBtnLeftHand.Clear;
  FNumAmmoLeft.Visible := False;
  if LeftHandItem <> nil then
  begin
    LeftHandItem.GetRules.DrawHandSprite(FMod.GetSurfaceSet('BIGOBS.PCK'), FBtnLeftHand);
    if (LeftHandItem.GetRules.GetBattleType = BT_FIREARM) and
       (LeftHandItem.NeedsAmmo or (LeftHandItem.GetRules.GetClipSize > 0)) then
    begin
      FNumAmmoLeft.Visible := True;
      if LeftHandItem.GetAmmoItem <> nil then
        FNumAmmoLeft.Value := LeftHandItem.GetAmmoItem.AmmoQuantity
      else
        FNumAmmoLeft.Value := 0;
    end;
  end;

  RightHandItem := GetRightHandItem(Unit);
  FBtnRightHand.Clear;
  FNumAmmoRight.Visible := False;
  if RightHandItem <> nil then
  begin
    RightHandItem.GetRules.DrawHandSprite(FMod.GetSurfaceSet('BIGOBS.PCK'), FBtnRightHand);
    if (RightHandItem.GetRules.GetBattleType = BT_FIREARM) and
       (RightHandItem.NeedsAmmo or (RightHandItem.GetRules.GetClipSize > 0)) then
    begin
      FNumAmmoRight.Visible := True;
      if RightHandItem.GetAmmoItem <> nil then
        FNumAmmoRight.Value := RightHandItem.GetAmmoItem.AmmoQuantity
      else
        FNumAmmoRight.Value := 0;
    end;
  end;

  if CheckFOV then
    FSave.GetTileEngine.CalculateFOV(Unit);

  J := 0;
  for var VisibleUnit in Unit.GetVisibleUnits do
    if J < VISIBLE_MAX then
    begin
      FBtnVisibleUnit[J].Visible := True;
      FNumVisibleUnit[J].Visible := True;
      FVisibleUnit[J] := VisibleUnit;
      Inc(J);
    end;

  ShowPsiButton(Unit.GetSpecialWeapon(BT_PSIAMP) <> nil);
end;

function TBattlescapeState.GetLeftHandItem(AUnit: TBattleUnit): TBattleItem;
var
  Melee: TBattleItem;
begin
  Melee := GetSpecialMeleeWeapon(AUnit);
  var LeftHand := AUnit.GetItem('STR_LEFT_HAND');
  if (Melee <> nil) and (AUnit.GetItem('STR_RIGHT_HAND') <> nil) and (LeftHand = nil) then
    Result := Melee
  else
    Result := LeftHand;
end;

function TBattlescapeState.GetRightHandItem(AUnit: TBattleUnit): TBattleItem;
var
  Melee: TBattleItem;
begin
  Melee := GetSpecialMeleeWeapon(AUnit);
  var RightHand := AUnit.GetItem('STR_RIGHT_HAND');
  if (Melee <> nil) and (RightHand = nil) then
    Result := Melee
  else
    Result := RightHand;
end;

function TBattlescapeState.GetSpecialMeleeWeapon(AUnit: TBattleUnit): TBattleItem;
begin
  Result := AUnit.GetSpecialWeapon(BT_MELEE);
end;

procedure TBattlescapeState.BlinkVisibleUnitButtons;
var
  Delta: Integer;
  Color: Byte;
  I: Integer;
begin
  Delta := 1; Color := 32;
  for I := 0 to VISIBLE_MAX - 1 do
    if FBtnVisibleUnit[I].Visible then
    begin
      FBtnVisibleUnit[I].DrawRect(0,0,15,12,15);
      FBtnVisibleUnit[I].DrawRect(1,1,13,10,Color);
    end;
  if Color = 44 then Delta := -2;
  if Color = 32 then Delta := 1;
  Color := Color + Delta;
end;

procedure TBattlescapeState.BlinkHealthBar;
var
  Color: Byte;
  Step: Integer;
  Bu: TBattleUnit;
  I: Integer;
begin
  Color := 0; Step := 0;
  Step := 1 - Step;
  Bu := FSave.GetSelectedUnit;
  if (Step = 0) or (Bu = nil) or (not FBarHealth.Visible) then Exit;
  Color := Color + 1;
  if Color > 3 then Color := 0;
  for I := 0 to 5 do
    if Bu.GetFatalWound(I) > 0 then
    begin
      FBarHealth.Color := FBarHealthColor + Color;
      Exit;
    end;
  if FBarHealth.Color <> FBarHealthColor then
    FBarHealth.Color := FBarHealthColor;
end;

procedure TBattlescapeState.FinishBattle(Abort: Boolean; InExitArea: Integer);
var
  Deploy: TAlienDeployment;
  NextStage: string;
  BGen: TBattlescapeGenerator;
  Cutscene: string;
begin
  while not FGame.IsState(Self) do FGame.PopState;
  FGame.GetCursor.Visible := True;
  if FSave.GetAmbientSound <> -1 then
    FMod.GetSoundByDepth(0, FSave.GetAmbientSound).StopLoop;
  Deploy := FMod.GetDeployment(FSave.GetMissionType);
  if Deploy = nil then
  begin
    for var Ufo in FGame.GetSavedGame.GetUfos do
      if Ufo.IsInBattlescape then
      begin
        Deploy := FMod.GetDeployment(Ufo.GetRules.GetType);
        Break;
      end;
  end;
  if Deploy <> nil then NextStage := Deploy.GetNextStage;
  if (not NextStage.IsEmpty) and (InExitArea > 0) then
  begin
    FPopups.Clear;
    FSave.MissionType := NextStage;
    BGen := TBattlescapeGenerator.Create(FGame);
    try
      BGen.NextStage;
      FGame.PopState;
      FGame.PushState(TBriefingState.Create(nil, nil));
    finally
      BGen.Free;
    end;
  end
  else
  begin
    FPopups.Clear;
    FAnimTimer.Stop;
    FGameTimer.Stop;
    FGame.PopState;
    FGame.PushState(TDebriefingState.Create);
    if Deploy <> nil then
    begin
      if Abort then Cutscene := Deploy.GetAbortCutscene
      else if InExitArea = 0 then Cutscene := Deploy.GetLoseCutscene
      else Cutscene := Deploy.GetWinCutscene;
    end;
    if not Cutscene.IsEmpty then
    begin
      FGame.PushState(TCutsceneState.Create(Cutscene));
      if Cutscene = CutsceneState.WIN_GAME then FGame.GetSavedGame.Ending := END_WIN;
      if Cutscene = CutsceneState.LOSE_GAME then FGame.GetSavedGame.Ending := END_LOSE;
      if (FGame.GetSavedGame.Ending <> END_NONE) and FGame.GetSavedGame.IsIronman then
        FGame.PushState(TSaveGameState.Create(OPT_BATTLESCAPE, SAVE_IRONMAN, FPalette));
    end;
  end;
end;

procedure TBattlescapeState.StopScrolling(AAction: TAction);
begin

  begin
    SDL_WarpMouse(FXBeforeMouseScrolling, FYBeforeMouseScrolling);
    AAction.SetMouseAction(FXBeforeMouseScrolling, FYBeforeMouseScrolling, FMap.X, FMap.Y);
    FBattleGame.SetupCursor;

       ((FSave.GetSide = FACTION_PLAYER) or FSave.GetDebugMode) then

    FMap.SetSelectorPosition(AAction.GetAbsoluteXMouse, AAction.GetAbsoluteYMouse);
  end;
