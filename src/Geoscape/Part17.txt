unit UfoDetectedState;

interface

uses
  System.SysUtils, System.Classes,
  Engine.State, Engine.Game, Engine.Mod, Engine.LocalizedText,
  Interface.TextButton, Interface.Window, Interface.Text,
  Interface.TextList, Savegame.Ufo, Mod.RuleUfo,
  Geoscape.GeoscapeState, Geoscape.Globe, Savegame.SavedGame,
  Engine.Options, Engine.Unicode, Savegame.AlienMission,
  Geoscape.InterceptState, Mod.RuleCraft;

type
  TUfoDetectedState = class(TState)
  private
    FUfo: TUfo;
    FState: TGeoscapeState;
    FBtnIntercept, FBtnCentre, FBtnCancel: TTextButton;
    FWindow: TWindow;
    FTxtUfo, FTxtDetected, FTxtHyperwave: TText;
    FLstInfo, FLstInfo2: TTextList;
    procedure BtnInterceptClick(AAction: TAction);
    procedure BtnCentreClick(AAction: TAction);
    procedure BtnCancelClick(AAction: TAction);
  public
    constructor Create(AUfo: TUfo; AState: TGeoscapeState; ADetected, AHyperwave: Boolean);
    destructor Destroy; override;
  end;

implementation

{ TUfoDetectedState }

constructor TUfoDetectedState.Create(AUfo: TUfo; AState: TGeoscapeState; ADetected, AHyperwave: Boolean);
var
  ss: string;
  altitude: string;
  heading: string;
  underwater: Boolean;
begin
  inherited Create(nil);
  FUfo := AUfo;
  FState := AState;

  // Generate IDs if needed
  if FUfo.Id = 0 then
    FUfo.Id := Game.SavedGame.GetId('STR_UFO');
  if (FUfo.Altitude = 'STR_GROUND') and (FUfo.LandId = 0) then
    FUfo.LandId := Game.SavedGame.GetId('STR_LANDING_SITE');

  FScreen := False;

  if AHyperwave then
    FWindow := TWindow.Create(Self, 224, 180, 16, 10, POPUP_BOTH)
  else
    FWindow := TWindow.Create(Self, 224, 128, 16, 44, POPUP_BOTH);

  FBtnIntercept := TTextButton.Create(200, 12, 28, 118);
  FBtnCentre := TTextButton.Create(200, 12, 28, 134);
  FBtnCancel := TTextButton.Create(200, 12, 28, 150);
  FTxtUfo := TText.Create(207, 17, 28, 53);
  FTxtDetected := TText.Create(100, 9, 28, 69);
  FTxtHyperwave := TText.Create(214, 17, 21, 44);
  FLstInfo := TTextList.Create(217, 32, 28, 80);
  FLstInfo2 := TTextList.Create(217, 32, 28, 96);

  if AHyperwave then
  begin
    FBtnIntercept.Y := 136;
    FBtnCentre.Y := 152;
    FBtnCancel.Y := 168;
    FTxtUfo.Y := 20;
    FTxtDetected.Y := 36;
    FLstInfo.Y := 60;
  end
  else
  begin
    FTxtHyperwave.Visible := False;
    FLstInfo2.Visible := False;
  end;

  SetInterface('UFOInfo', AHyperwave);

  Add(FWindow, 'window', 'UFOInfo');
  Add(FBtnIntercept, 'button', 'UFOInfo');
  Add(FBtnCentre, 'button', 'UFOInfo');
  Add(FBtnCancel, 'button', 'UFOInfo');
  Add(FTxtUfo, 'text', 'UFOInfo');
  Add(FTxtDetected, 'text', 'UFOInfo');
  Add(FTxtHyperwave, 'text', 'UFOInfo');
  Add(FLstInfo, 'text', 'UFOInfo');
  Add(FLstInfo2, 'text', 'UFOInfo');

  CenterAllSurfaces;

  FWindow.SetBackground(Game.Mod.GetSurface('BACK15.SCR'));

  FBtnIntercept.Text := Tr('STR_INTERCEPT');
  FBtnIntercept.OnMouseClick := BtnInterceptClick;

  FBtnCentre.Text := Tr('STR_CENTER_ON_UFO_TIME_5_SECONDS');
  FBtnCentre.OnMouseClick := BtnCentreClick;

  FBtnCancel.Text := Tr('STR_CANCEL_UC');
  FBtnCancel.OnMouseClick := BtnCancelClick;
  FBtnCancel.OnKeyboardPress(Options.KeyCancel, BtnCancelClick);

  if ADetected then
    FTxtDetected.Text := Tr('STR_DETECTED')
  else
    FTxtDetected.Text := '';

  FTxtHyperwave.Align := ALIGN_CENTER;
  FTxtHyperwave.WordWrap := True;
  FTxtHyperwave.Text := Tr('STR_HYPER_WAVE_TRANSMISSIONS_ARE_DECODED');

  FTxtUfo.Big := True;
  FTxtUfo.Text := FUfo.Name(Game.Language);

  // Info list 1
  FLstInfo.SetColumns(2, 77, 140);
  FLstInfo.Dot := True;

  ss := Unicode.TOK_COLOR_FLIP + Tr(FUfo.Rules.Size);
  FLstInfo.AddRow(2, [Tr('STR_SIZE_UC'), ss]);

  altitude := FUfo.Altitude;
  if altitude = 'STR_GROUND' then altitude := 'STR_GROUNDED';
  // Check if underwater craft detection
  underwater := False;
  for var craftName in Game.Mod.CraftsList do
    if Game.Mod.GetCraft(craftName).IsWaterOnly then
    begin
      underwater := True;
      Break;
    end;
  if underwater and not FState.Globe.InsideLand(FUfo.Longitude, FUfo.Latitude) then
    altitude := 'STR_AIRBORNE';
  ss := Unicode.TOK_COLOR_FLIP + Tr(altitude);
  FLstInfo.AddRow(2, [Tr('STR_ALTITUDE'), ss]);

  heading := FUfo.Direction;
  if FUfo.Status <> UfoStatus.FLYING then
    heading := 'STR_NONE_UC';
  ss := Unicode.TOK_COLOR_FLIP + Tr(heading);
  FLstInfo.AddRow(2, [Tr('STR_HEADING'), ss]);

  ss := Unicode.TOK_COLOR_FLIP + Unicode.FormatNumber(FUfo.Speed);
  FLstInfo.AddRow(2, [Tr('STR_SPEED'), ss]);

  // Info list 2 (hyperwave)
  FLstInfo2.SetColumns(2, 77, 140);
  FLstInfo2.Dot := True;

  ss := Unicode.TOK_COLOR_FLIP + Tr(FUfo.Rules.TypeName);
  FLstInfo2.AddRow(2, [Tr('STR_CRAFT_TYPE'), ss]);

  ss := Unicode.TOK_COLOR_FLIP + Tr(FUfo.AlienRace);
  FLstInfo2.AddRow(2, [Tr('STR_RACE'), ss]);

  ss := Unicode.TOK_COLOR_FLIP + Tr(FUfo.MissionType);
  FLstInfo2.AddRow(2, [Tr('STR_MISSION'), ss]);

  ss := Unicode.TOK_COLOR_FLIP + Tr(FUfo.Mission.Region);
  FLstInfo2.AddRow(2, [Tr('STR_ZONE'), ss]);
end;

destructor TUfoDetectedState.Destroy;
begin
  inherited;
end;

procedure TUfoDetectedState.BtnInterceptClick(AAction: TAction);
begin
  FState.TimerReset;
  FState.Globe.Center(FUfo.Longitude, FUfo.Latitude);
  Game.PushState(TInterceptState.Create(FState.Globe, nil, FUfo));
end;

procedure TUfoDetectedState.BtnCentreClick(AAction: TAction);
begin
  FState.TimerReset;
  FState.Globe.Center(FUfo.Longitude, FUfo.Latitude);
  Game.PopState;
end;

procedure TUfoDetectedState.BtnCancelClick(AAction: TAction);
begin
  Game.PopState;
end;

end.

uses

{ TUfoLostState }

constructor TUfoLostState.Create(const AId: string);
begin
  inherited Create(nil);
  FId := AId;
  FScreen := False;

  FWindow.SetBackground(Game.Mod.GetSurface('BACK15.SCR'));

  FTxtTitle.Big := True;
  FTxtTitle.Align := ALIGN_CENTER;

procedure TUfoLostState.BtnOkClick(AAction: TAction);

end.
